 function market(season,yr){

   if(f.st.food>reserve){ const s=(f.st.food-reserve)*0.5; f.st.food-=s; f.savings+=s*price.food*0.9; marketFood+=s; }

  if(R()<0.22*H){const v=sev();HZ.winter=1+0.35*v;HZ.log.push('harsh winter '+v.toFixed(1));}

  if(yr>=1&&R()<pr){ const t=R(); const ph=0.1*Math.exp(-A.deter/4); const tier=t<ph?['horde',80,150]:t<ph+0.3?['warband',30,65]:['band',12,28];

 function raid(yr){ if(HZ.raidS)battle(HZ.raidS,'raid ('+HZ.raidTier+')',false); if(HZ.war)battle(HZ.war,'war vs '+A.warWith.name,true); }
 function battle(R0,label,isWar){ const D=defense()+(isWar?A.allyHelp:A.allyHelp*0.3);
  if(D>=R0){ A.deter+=isWar?0.5:1; if(isWar&&R()<0.5){A.war=0;} const cas=Math.min(A.soldiers,Math.round(R0*0.04)); A.soldiers-=cas; deaths+=cas; HZ.log.push(label+' repelled '+Math.round(R0)); A.lastRaid='won'; return; }
  A.deter=Math.max(0,A.deter-1);
  const loss=Math.min(0.75,0.25+(R0-D)/R0*0.5); T.food*=1-loss; T.goods*=1-loss; T.silver*=1-loss*0.8; oreStock*=1-loss*0.5;
  const burned=Math.floor(fams.length*loss*0.15); for(let i=0;i<burned;i++){const f=fams[Math.floor(R()*fams.length)]; f.st.food*=0.2; f.savings*=0.3; f.happy-=10;}
  const killed=killPeople(Math.round((R0-D)*0.22)); const cas=Math.round(A.soldiers*0.3); A.soldiers-=cas; A.mercs=Math.round(A.mercs*0.6); deaths+=cas;
  // disloyal mercs: some switch sides/flee after a loss
  if(A.mercs>0&&R()<0.5){A.mercs=Math.round(A.mercs*0.5);HZ.log.push('mercenaries fled');}
  for(const f of fams)f.happy-=8*loss; sackedYrs++; A.lastRaid='lost'; HZ.log.push((isWar?'WAR LOST ':'RAID SACKED ')+label+' '+Math.round(R0)+' vs '+Math.round(D)+' killed '+killed); if(isWar){ const lostF=Math.floor(fams.length*0.06); fams.splice(0,lostF); HZ.log.push('ceded land: '+lostF+' steads'); A.war=0; } }
 function plague(yr){ const hungry=fams.filter(f=>f.starve>0).length/Math.max(1,fams.length); const frac=Math.min(0.3,(0.04+0.1*HZ.plague)*(1+hungry)*(townPop()>P.townCap*0.9?1.4:1));
  const pop=fams.reduce((a,f)=>a+f.adults+f.kids,0); const k=killPeople(Math.round(pop*frac)); for(const c of crafters)if(R()<frac)c.size=Math.max(1,c.size-1); A.soldiers=Math.round(A.soldiers*(1-frac)); HZ.log.push('plague killed '+k); }
 function fire(yr){ const prot=[1,0.8,0.6,0.45][Math.min(3,T.stLv)]; const l=Math.min(0.5,0.12*HZ.fire*prot*(1.4-0.8*P.layout)); T.goods*=1-l; T.food*=1-l*0.7; for(const c of crafters)if(R()<0.15*HZ.fire){c.stock=0;c.silver*=0.6;} HZ.log.push('fire −'+Math.round(l*100)+'% stores'); }

 function D_low(){return defense()+A.allyHelp*0.5;}
 function adapt(yr,last){ if(!P.adaptive||!last)return; P.tactics=0.35+0.4*P.skill; if(R()>P.skill)return; const pop=last.pop; const needYr=pop*30;
  const foodCover=(T.food+fams.reduce((a,f)=>a+f.st.food,0))/Math.max(1,needYr);
  if(last.happy<45&&P.tithe>0.12)P.tithe-=0.05; else if(foodCover<0.6&&P.tithe<0.3)P.tithe+=0.05; else if(foodCover>1.5&&P.tithe>0.15)P.tithe-=0.05;
  const rivMax=Math.max(0,...RIV.map(r=>r.str)); const threat=Math.max(45*Math.exp(-A.deter/6)*(1+Math.sqrt(T.goods+T.silver)*0.01), rivMax*0.9)*P.adaptThreat/1.3;
  if(D_low()<threat&&P.wallLevel<3&&T.goods+T.silver>900*(A.walls+1))P.wallLevel=A.walls+1;
  if(rivMax>defense()*1.1)P.tribute=Math.min(200,P.tribute+50); else if(rivMax<defense()*0.8)P.tribute=Math.max(0,P.tribute-50); const D=defense(); const sacked=last.hazards&&last.hazards.some(x=>/SACKED/.test(x));
  if(D<threat||sacked){ const gap=Math.max(0.2,(threat-D)/Math.max(1,threat)); if(foodCover>0.5&&last.happy>35)P.armyTarget=Math.min(Math.round(pop*0.09),P.armyTarget+Math.round(4+12*gap)); if(T.silver>300)P.mercs=Math.min(70,P.mercs+Math.round(3+10*gap)); }
  if(P.armyTarget>pop*0.1)P.armyTarget=Math.round(pop*0.1);
  if(sacked&&P.storageLevel<3)P.storageLevel++;
  else if(D>threat*1.6){ P.armyTarget=Math.max(10,P.armyTarget-5); P.mercs=Math.max(0,P.mercs-5); }
  if(last.wasted>T.food*0.2&&P.storageLevel<3)P.storageLevel++;
  P.civic=last.happy<45?0.5:0.2;
 }
 function yearEnd(yr){
  const brokeYr={};
  // births / deaths / happiness
  const civicHappy=Math.min(15, A.prestige/Math.max(1,fams.length)*0.6);
 for(const f of [...fams]){
   const n=f.adults+f.kids; const fed=f.starve<5;
   const sav=f.savings/(n*10);
   f.happy+= (60 + civicHappy - P.tithe*90 + Math.min(20,sav*8) + (f.starve>0?-Math.min(30,f.starve):0) + (f.cart?3:0) - f.happy)*0.5;
   f.happy=Math.max(0,Math.min(100,f.happy));
   if(fed&&R()<0.38+(f.happy-50)/250) {f.kids++;births++;}
   if(f.kids>0&&R()<0.22){f.kids--;f.adults++;}
   const dp=0.05+f.starve*0.012; if(R()<dp*n/5){ if(f.adults>1&&R()<0.6)f.adults--;else if(f.kids>0)f.kids--; deaths++; if(f.starve>5)deathsStarve++; }
   f.starve=0;
   // cart / donkey purchase or homemade
   if(f.cart<2){ const cost=f.cart===0?45:120; if(f.savings>cost*1.6){f.savings-=cost;f.cart++;} else if(f.cart===0&&f.st.wood>12&&R()<0.15){f.st.wood-=12;f.cart=1;} }
   // savings decay (spoilage/repairs)
   f.savings*=0.92;
   // family split to new stead
   if(f.adults+f.kids>=10&&fams.length<P.maxStead){ const nf=newFamily(0); nf.adults=2;nf.kids=1; f.adults-=2;f.kids-=1; nf.st.food=Math.min(f.st.food*0.3,120); f.st.food-=nf.st.food; fams.push(nf); splits++; }
   // break-off to crafter
   const k=f.craft;
   const ready=f.skill>=1&&f.savings>=P.breakSavings*(1-civicHappy/50)&&demand[k]>=P.breakDemand&&f.adults>=3;
   if(ready&&townPop()+2>P.townCap){blocked++;}
   else if(ready&&!brokeYr[k]){ brokeYr[k]=1;
     f.adults-=1; f.savings-=P.breakSavings*0.8; f.skill=R()*0.3; leftForCraft++;
     const c={type:k,size:1+(R()<0.5?1:0),skill:1,silver:P.breakSavings*0.5,stock:0,food:30,hungry:0,born:yr,earnYr:0};
     if(c.size===2&&f.kids>0){f.kids--;} crafters.push(c); if(firstCrafter===null)firstCrafter=yr+1;
     ev.push({y:yr+1,t:'crafter',k}); demand[k]-=0.3;
   }
   if(f.adults<=0){ fams.splice(fams.indexOf(f),1); ev.push({y:yr+1,t:'stead abandoned'}); }
  }
  // crafters: upkeep, growth, promote
  for(const c of [...crafters]){
   c.silver*=0.95; if(c.earnYr>60&&c.size<5&&R()<0.25&&townPop()<P.townCap)c.size++;
   if(c.hungry>20&&R()<0.5){ crafters.splice(crafters.indexOf(c),1); ev.push({y:yr+1,t:'crafter failed',k:c.type}); continue; }
   c.hungry=0; c.earnYr=0;
   const elig=c.silver>=P.merchantSilver&&yr+1-c.born>=P.merchantYears&&R()<Math.min(1,Math.max(0.05,((P.foreignPrice*(1+P.regionalMargin)-1)*(1-P.mtax)-0.05)/0.12));
   if(elig&&merchants.length>=P.charterMax){ c.denied=(c.denied||0)+1; deniedYr++; // charter refused
     if(c.denied>=2&&R()<0.35){ crafters.splice(crafters.indexOf(c),1); ev.push({y:yr+1,t:'denied crafter left with silver',k:c.type}); continue; } }
   else if(elig){
     crafters.splice(crafters.indexOf(c),1); merchants.push({from:c.type,size:c.size+1,silver:c.silver,born:yr}); if(firstMerchant===null)firstMerchant=yr+1; ev.push({y:yr+1,t:'merchant',k:c.type}); }
  }
  // merchants: buy raw goods from holding, regional trade, tax, license fees
  for(const m of [...merchants]){
   const buy=0; // merchants now buy holding surplus via the sell-surplus policy (cargo)
   const capital=m.silver+buy; // STUB regional trade: export up to regionalCap per house at fixed foreign price
   const ex=Math.min(m.cargo||0,P.regionalCap); m.cargo=((m.cargo||0)-ex)*0.9; const rev=ex*0.8*P.foreignPrice*(1+P.regionalMargin); const own=Math.min(m.silver,P.regionalCap*0.5)*(P.foreignPrice*(1+P.regionalMargin)-1)*0.5; const profit=rev+own; m.tradeIn=rev+own; m.silver-=0; 
   const tax=Math.max(0,profit-ex*0.8)*P.mtax; m.silver+=profit-tax; T.silver+=tax*0.6; T.goods+=tax*0.4; m.taxYr=tax;
   
   m.silver*=0.96;
   // heavy tax drives merchant houses to move their trade elsewhere
   if(P.mtax>0.3&&R()<(P.mtax-0.3)*0.8){ merchants.splice(merchants.indexOf(m),1); ev.push({y:yr+1,t:'merchant left region'}); }
  }
  silverIn=merchants.reduce((a,m)=>a+(m.tradeIn||0),0); merchTaxYr=merchants.reduce((a,m)=>a+(m.taxYr||0),0);
  // ---------- Holding spending policies (tithe sinks)
  const pay=(v,cat)=>{ let left=v; const g=Math.min(T.goods,left); T.goods-=g; left-=g; const sv=Math.min(T.silver,left); T.silver-=sv; left-=sv; const fd=Math.min(T.food*0.5,left); T.food-=fd; left-=fd; A.spend[cat]+=v-left; return left<=1e-6; };
  for(const k in A.spend)A.spend[k]=0;
  pay(P.household,'household');
  if(P.armyTarget>0&&!A.tg&&T.goods+T.silver>500){ pay(400,'army'); A.tg=true; }
  if(A.tg)pay(40,'army');
  if(A.soldiers>0&&!pay(A.soldiers*6,'army')){ const d=Math.ceil(A.soldiers*0.2); A.soldiers-=d; A.unpaid++; ev.push({y:yr+1,t:'unpaid soldiers deserted',k:d}); }
  else A.unpaid=0;
  let recruits=Math.min(P.armyTarget-A.soldiers, Math.ceil(P.armyTarget*0.25)+3);
  const pool=fams.filter(f=>f.adults>=3).sort((a,b)=>b.adults-a.adults);
  for(let i=0;i<recruits&&i<pool.length;i++){ if(!pay(20,'army'))break; pool[i].adults--; pool[i].happy-=3; A.soldiers++; }
  if(A.soldiers>P.armyTarget){ const x=A.soldiers-P.armyTarget; A.soldiers-=x; const fs2=fams.slice().sort((a,b)=>a.adults-b.adults); for(let i=0;i<x;i++)fs2[i%fs2.length].adults++; }
  A.arms*=0.9; A.armor*=0.92; { const o2=Math.min(oreStock*0.5,A.soldiers*2); oreStock-=o2; A.arms+=o2*6; A.spend.army+=o2*6; const o3=Math.min(oreStock*0.5,A.soldiers*1.5); oreStock-=o3; A.armor+=o3*4; const o4=oreStock*P.monumentOre; oreStock-=o4; A.prestige+=o4*0.3; } // royal armoury turns ore into weapons
  for(const m of MINES){ if(mineLv[m]<P.mineLevel&&T.goods+T.silver>350*(mineLv[m]+1)*1.5){ if(pay(350*(mineLv[m]+1),'mines'))mineLv[m]++; } }
  // Holding buys refined goods from crafters at a 1.5x premium (patronage)
  if(A.walls<P.wallLevel&&T.goods+T.silver+T.food*0.3>700*(A.walls+1)){ if(pay(700*(A.walls+1),'walls'))A.walls++; }
  if(A.walls>0&&!pay(40*A.walls,'walls')&&R()<0.3){A.walls--; ev.push({y:yr+1,t:'walls fell into disrepair'});}
  A.allyHelp=Math.min(60,A.relations*6);
  A.deter=A.deter*0.95+(defense()>40?0.3:0);
  if(T.stLv<P.storageLevel&&pay(400*(T.stLv+1),'storage'))T.stLv++; pay(30*T.stLv,'storage');

  let strength=defense();
  // ---------- silver sinks
  { const reserve=300+A.mercs*20; let free=Math.max(0,T.silver-reserve);
    const paySilver=(v,cat)=>{const x=Math.min(T.silver,v);T.silver-=x;A.spend[cat]+=x;return x>=v-1e-6;};
    // mercenaries: silver army, no draft
    if(A.mercs>0&&!paySilver(A.mercs*15*(1+A.mercs/50),'mercs')){ A.mercLoyal=Math.max(0.3,A.mercLoyal-0.3); const d=Math.ceil(A.mercs*0.5); A.mercs-=d; A.mercUnpaid=(A.mercUnpaid||0)+1; ev.push({y:yr+1,t:'unpaid mercenaries left (pillaged)',k:d}); T.goods*=0.85; for(const f of fams)f.happy-=4; } else { A.mercUnpaid=0; A.mercLoyal=Math.min(0.75,A.mercLoyal+0.1); }
    let hired=0; const sup=Math.round(P.mercSupply*(0.5+R())); while(A.mercs<P.mercs&&hired<sup&&T.silver>reserve+30*(1+A.mercs/25)){ paySilver(30*(1+A.mercs/25),'mercs'); A.mercs++; hired++; }
    if(A.mercs>P.mercs)A.mercs=P.mercs;
    free=Math.max(0,T.silver-reserve);
    // parks, fountains, temples, palaces, festivals -> prestige (decays 10%/yr)
    const civ=free*P.civic; paySilver(civ,'civic'); A.prestige=A.prestige*0.9+civ/20;
    // imperial building upgrades -> +output/storage (diminishing, max +100%)
    const up=Math.max(0,T.silver-reserve)*P.upgrades; paySilver(up,'upgrades'); A.upg=Math.min(1,A.upg+up/(4000*(1+A.upg*3)));
    // foreign relations STUB: tribute/gifts only move a meter
    if(P.tribute>0){ const ok=paySilver(P.tribute,'tribute')||pay(P.tribute*0.5,'tribute'); A.relations=A.relations*0.85+(ok?P.tribute/50:P.tribute/120); } else A.relations*=0.85;
  }

  const pop=fams.reduce((a,f)=>a+f.adults+f.kids,0)+crafters.reduce((a,c)=>a+c.size,0)+merchants.reduce((a,m)=>a+m.size,0)+A.soldiers;

  { const prev=hist[hist.length-2]; const pe=[];
    if(HZ.log.some(x=>/drought|poor harvest/.test(x)))pe.push('crop failure'); if(HZ.winter>1)pe.push('harsh winter'); if(HZ.raidS)pe.push(HZ.log.some(x=>/SACKED/.test(x))?'raid lost':'raid'); if(HZ.war)pe.push(HZ.log.some(x=>/WAR LOST/.test(x))?'war lost':'war'); if(HZ.plague)pe.push('plague'); if(HZ.fire)pe.push('fire');
    if(prev&&L.starveDeaths>prev.starveDeaths)pe.push('food shortage'); if(L.holdingShort>0)pe.push('holding short');
    if(L.happy<45)pe.push('unrest'); if(L.wasted>Math.max(300,(L.tFood+L.tGoods)*0.15))pe.push('spoilage'); if(L.blocked>0||fams.filter(f=>f.adults<2).length>fams.length*0.15)pe.push('labour gap');
    if(L.unpaid>0||L.mercUnpaid>0)pe.push('pay arrears'); if(L.oreStock>400)pe.push('ore glut'); if(L.denied>0)pe.push('charter petitions');
    L.pressure=pe; HZ.famine=pe.includes("food shortage"); }
  peakPop=Math.max(peakPop,pop); lowHappyYrs=L.happy<25?lowHappyYrs+1:0;
  if(pop<peakPop*0.4||pop<70) dead={y:yr+1,cause:sackedYrs>=2?'raids/war':(L.starveDeaths>pop*0.3?'famine':'depopulation')};
  else if(lowHappyYrs>=3) dead={y:yr+1,cause:'revolt'};
  else if(sackedYrs>=P.sackLimit) dead={y:yr+1,cause:'raids/war'};
  if(dead){ev.push({y:dead.y,t:'CITY FELL: '+dead.cause});}
  sackedYrs=Math.max(0,sackedYrs-(A.lastRaid==='won'?1:0.34)); A.lastRaid='';
  adapt(yr,L);
  stats.deliveryDays=0; oreYr=0; T.short=0; T.wasted=0; T.soldYr=0; blocked=0; deniedYr=0;
 }
 return {hist,ev,dead,firstCrafter,firstMerchant,diagnose:(dead?['FELL yr '+dead.y+': '+dead.cause]:[]).concat(diagnose(hist,P,firstCrafter,firstMerchant))};
}
function diagnose(h,P,fc,fm){
 const last=h[h.length-1],first=h[0]; const F=[];
 if(fc===null||last.crafters+last.merchants<2)F.push('stagnation');
 if(last.merchWealth>Math.max(2000,last.treasury*1.0))F.push('runaway merchants');
 const p5=h[Math.max(0,h.length-6)]; if(last.starveDeaths-p5.starveDeaths>last.pop*0.03||last.pop<first.pop*0.7)F.push('starvation spiral');
 if(last.families<first.families*0.8)F.push('crafter exodus');
 const tail=h.slice(-5); if(tail.reduce((a,x)=>a+x.holdingShort,0)>20||last.treasury<50)F.push('treasury collapse');
 if(h.slice(-5).reduce((a,x)=>a+x.wasted,0)>h.slice(-5).reduce((a,x)=>a+x.treasury,0)*0.3)F.push('heavy spoilage');
 if(last.oreStock>400)F.push('ore glut');
 if(h.slice(-5).some(x=>x.unpaid>0))F.push('army unpaid/desertion');
 if(h.slice(-5).some(x=>x.mercUnpaid>0))F.push('mercenary pillage');
 if(last.tSilver>last.pop*60)F.push('silver pile-up');
 if(last.happy<35)F.push('unrest');
 if(h.slice(-5).reduce((a,x)=>a+x.blocked,0)>10)F.push('town housing cap');
 return F;
}
